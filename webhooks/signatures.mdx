Every webhook request from KyrenPay includes a cryptographic signature that you should verify to ensure the request is authentic and hasn't been tampered with.

## Signature Algorithm

KyrenPay uses **HMAC-SHA256** to sign webhook payloads:

```
signature = "sha256=" + HMAC-SHA256(timestamp + "." + payload, webhook_secret)
```

The signature and timestamp are sent in HTTP headers:

| Header | Example |
| --- | --- |
| `X-Kyren-Signature` | `sha256=5d2dc4f1a...` |
| `X-Kyren-Timestamp` | `1704628800000` (Unix timestamp in milliseconds) |

## Verification Steps

<Steps>
  <Step title="Extract headers">
    Read `X-Kyren-Signature` and `X-Kyren-Timestamp` from the request headers.
  </Step>
  <Step title="Check timestamp">
    Verify the timestamp is within 5 minutes (300000 milliseconds) of the current time. This prevents replay attacks.
  </Step>
  <Step title="Compute expected signature">
    Concatenate `timestamp + "." + raw_request_body`, then compute HMAC-SHA256 using your webhook secret.
  </Step>
  <Step title="Compare signatures">
    Compare the computed signature with the one in the header. Use constant-time comparison to prevent timing attacks.
  </Step>
</Steps>

## Code Examples

Register the raw Webhook route before any global JSON parser, or parse JSON only on regular routes. Verify headers and the original body before parsing the event. The examples below require your durable inbox adapter: uniquely save an event ID and pending payload before returning `2xx`; a worker must retry and fulfill idempotently. Invalid headers return a rejection, not an unhandled exception. Each retry uses a fresh sending timestamp; use the header, not `created_at`, for signature verification.

### Node.js

```javascript
const crypto = require('crypto');

function verifyWebhookSignature(payload, signature, timestamp, secret) {
  if (typeof secret !== 'string' || !secret) return false;
  if (typeof signature !== 'string' || !/^sha256=[0-9a-f]{64}$/.test(signature)) return false;
  if (typeof timestamp !== 'string' || !/^\d+$/.test(timestamp)) return false;
  const sentAt = Number(timestamp);
  if (!Number.isSafeInteger(sentAt) || Math.abs(Date.now() - sentAt) > 300000) return false;
  const data = `${timestamp}.${payload}`;
  const expected = 'sha256=' + crypto
    .createHmac('sha256', secret)
    .update(data)
    .digest('hex');

  return crypto.timingSafeEqual(
    Buffer.from(expected),
    Buffer.from(signature)
  );
}

app.post('/webhooks/kyren',
  express.raw({ type: 'application/json' }),
  async (req, res) => {
    const signature = req.headers['x-kyren-signature'];
    const timestamp = req.headers['x-kyren-timestamp'];
    if (!Buffer.isBuffer(req.body)) return res.status(400).send('Raw JSON body required');
    const payload = req.body.toString('utf8');

    if (!verifyWebhookSignature(payload, signature, timestamp, process.env.KYREN_WEBHOOK_SECRET)) {
      return res.status(400).send('Invalid signature');
    }

    let event;
    try { event = JSON.parse(payload); } catch { return res.status(400).send('Invalid JSON'); }
    if (!event || typeof event.id !== 'string' || !event.id || typeof event.type !== 'string' || !event.data) {
      return res.status(400).send('Invalid event');
    }
    try {
      await webhookInbox.insertIfAbsent(event.id, event);
    } catch {
      return res.status(503).send('Retry later');
    }
    res.status(200).send('OK');
  }
);
```

### Python

```python
import hmac
import hashlib
import time
import re

def verify_webhook_signature(payload: str, signature: str, timestamp: str, secret: str) -> bool:
    # Check timestamp tolerance (5 minutes)
    if not isinstance(secret, str) or not secret:
        return False
    if not isinstance(signature, str) or not re.fullmatch(r"sha256=[0-9a-f]{64}", signature):
        return False
    if not isinstance(timestamp, str) or not re.fullmatch(r"[0-9]{1,16}", timestamp):
        return False
    if int(timestamp) > 9007199254740991:
        return False
    current_time = int(time.time() * 1000)
    if abs(current_time - int(timestamp)) > 300000:
        return False

    # Compute expected signature
    data = f"{timestamp}.{payload}"
    expected = "sha256=" + hmac.new(
        secret.encode("utf-8"),
        data.encode("utf-8"),
        hashlib.sha256
    ).hexdigest()

    # Constant-time comparison
    return hmac.compare_digest(expected, signature)


# Usage in Flask
from flask import Flask, request

app = Flask(__name__)

@app.route("/webhooks/kyren", methods=["POST"])
def webhook():
    payload = request.get_data(as_text=True)
    signature = request.headers.get("X-Kyren-Signature", "")
    timestamp = request.headers.get("X-Kyren-Timestamp", "")

    if not verify_webhook_signature(payload, signature, timestamp, WEBHOOK_SECRET):
        return "Invalid signature", 400

    event = request.get_json()
    if not isinstance(event, dict) or not isinstance(event.get("id"), str) or not event["id"]:
        return "Invalid event", 400
    try:
        webhook_inbox.insert_if_absent(event["id"], event)
    except Exception:
        return "Retry later", 503

    return "OK", 200
```

### Go

```go
package main

import (
	"crypto/hmac"
	"crypto/sha256"
	"encoding/hex"
	"fmt"
	"io"
	"encoding/json"
	"math"
	"net/http"
	"strconv"
	"regexp"
	"time"
)

func verifyWebhookSignature(payload, signature, timestamp, secret string) bool {
    if secret == "" || !regexp.MustCompile("^sha256=[0-9a-f]{64}$").MatchString(signature) ||
        !regexp.MustCompile("^[0-9]+$").MatchString(timestamp) {
        return false
    }
	// Check timestamp tolerance (5 minutes)
	ts, err := strconv.ParseInt(timestamp, 10, 64)
	if err != nil || ts > 9007199254740991 {
		return false
	}
	currentTime := time.Now().UnixMilli()
	if math.Abs(float64(currentTime-ts)) > 300000 {
		return false
	}

	// Compute expected signature
	data := fmt.Sprintf("%s.%s", timestamp, payload)
	mac := hmac.New(sha256.New, []byte(secret))
	mac.Write([]byte(data))
	expected := "sha256=" + hex.EncodeToString(mac.Sum(nil))

	// Constant-time comparison
	return hmac.Equal([]byte(expected), []byte(signature))
}

func webhookHandler(w http.ResponseWriter, r *http.Request) {
    body, err := io.ReadAll(r.Body)
    if err != nil {
        http.Error(w, "Invalid body", http.StatusBadRequest)
        return
    }
	payload := string(body)
	signature := r.Header.Get("X-Kyren-Signature")
	timestamp := r.Header.Get("X-Kyren-Timestamp")

	if !verifyWebhookSignature(payload, signature, timestamp, webhookSecret) {
		http.Error(w, "Invalid signature", http.StatusBadRequest)
		return
	}

    var event struct { ID string `json:"id"` }
    if json.Unmarshal(body, &event) != nil || event.ID == "" {
        http.Error(w, "Invalid event", http.StatusBadRequest)
        return
    }
    if err := persistWebhookIfAbsent(event.ID, body); err != nil {
        http.Error(w, "Retry later", http.StatusServiceUnavailable)
        return
    }
	w.WriteHeader(http.StatusOK)
	w.Write([]byte("OK"))
}
```

<Warning>
  **Important security considerations:**
  - Always verify signatures before processing events
  - Use constant-time comparison functions to prevent timing attacks
  - Reject requests with timestamps older than 5 minutes (300000 ms) to prevent replay attacks
  - Use the raw request body for signature verification (before JSON parsing)
</Warning>
